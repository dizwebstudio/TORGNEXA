package securitysettingsrepo

import (
	"context"
	"database/sql"
	"fmt"
	"os"
	"strings"
	"testing"
	"time"

	_ "github.com/jackc/pgx/v5/stdlib"
	"github.com/torgnexa/torgnexa/internal/core/tenancy"
	"github.com/torgnexa/torgnexa/internal/platform/securitysettings"
)

func TestAuditConcurrentFirstSessionObservation(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	db, err := sql.Open("pgx", "host=/tmp/torgnexa-audit-pg-socket user=postgres dbname=audit sslmode=disable")
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	setup := `CREATE TABLE workspaces(organization_id text,id text,PRIMARY KEY(organization_id,id));
INSERT INTO workspaces VALUES('018f0000-0000-7000-8000-000000000001','018f0000-0000-7000-8000-000000000002');`
	if _, err = db.ExecContext(ctx, setup); err != nil {
		t.Fatal(err)
	}
	migration, err := os.ReadFile("../../../../migrations_legacy_pre_v1/000061_settings_security.sql")
	if err != nil {
		t.Fatal(err)
	}
	ddl := strings.Split(string(migration), "INSERT INTO migration_history")[0] + "COMMIT;"
	if _, err = db.ExecContext(ctx, ddl); err != nil {
		t.Fatal(err)
	}
	// Widen the natural SELECT-missing / INSERT interleaving deterministically.
	if _, err = db.ExecContext(ctx, `CREATE FUNCTION audit_delay_insert() RETURNS trigger LANGUAGE plpgsql AS 'BEGIN PERFORM pg_sleep(0.5); RETURN NEW; END';
CREATE TRIGGER audit_delay BEFORE INSERT ON settings_identity_sessions FOR EACH ROW EXECUTE FUNCTION audit_delay_insert();
CREATE ROLE auditapp LOGIN;
GRANT SELECT,INSERT,UPDATE ON settings_identity_sessions TO auditapp;
GRANT SELECT,INSERT ON settings_login_events TO auditapp;`); err != nil {
		t.Fatal(err)
	}
	app, err := sql.Open("pgx", "host=/tmp/torgnexa-audit-pg-socket user=auditapp dbname=audit sslmode=disable")
	if err != nil {
		t.Fatal(err)
	}
	defer app.Close()
	app.SetMaxOpenConns(8)
	repo, err := New(app)
	if err != nil {
		t.Fatal(err)
	}
	scope, _ := tenancy.ParseScope("018f0000-0000-7000-8000-000000000001", "018f0000-0000-7000-8000-000000000002")
	now := time.Now().UTC()
	start := make(chan struct{})
	results := make(chan error, 8)
	for i := 0; i < 8; i++ {
		go func(i int) {
			<-start
			results <- repo.Observe(ctx, scope, securitysettings.Observation{EventID: fmt.Sprintf("018f0000-0000-7000-8000-%012d", i+10), SessionRef: strings.Repeat("a", 64), SubjectRef: strings.Repeat("b", 64), ClientKind: "browser", AuthenticatedAt: now, ObservedAt: now, ExpiresAt: now.Add(time.Hour)})
		}(i)
	}
	close(start)
	failures := 0
	for i := 0; i < 8; i++ {
		if err := <-results; err != nil {
			failures++
			t.Log(err)
		}
	}
	if failures > 0 {
		t.Fatalf("valid concurrent requests rejected: %d/8", failures)
	}
}
